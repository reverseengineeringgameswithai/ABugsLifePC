def load(p):
 text=p.read_text(); decoder=json.JSONDecoder(); documents=[]; offset=0
 while offset<len(text):
  while offset<len(text) and text[offset].isspace(): offset+=1
  if offset==len(text): break
  value,offset=decoder.raw_decode(text,offset); documents.append(value)
 if not documents: raise ValueError('empty JSON stream: '+str(p))
 return documents[0] if len(documents)==1 else {'stream_format':'concatenated JSON command output','documents':documents}

"""Brief scoped stop plus register inspection; no breakpoints or register writes."""
import ctypes,os,signal,time
from pathlib import Path
from contextlib import contextmanager
class Registers(ctypes.Structure):
 _fields_=[(name,ctypes.c_ulonglong) for name in 'r15 r14 r13 r12 rbp rbx r11 r10 r9 r8 rax rcx rdx rsi rdi orig_rax rip cs eflags rsp ss fs_base gs_base ds es fs gs'.split()]
libc=ctypes.CDLL(None,use_errno=True);libc.ptrace.restype=ctypes.c_long
libc.ptrace.argtypes=[ctypes.c_ulong,ctypes.c_ulong,ctypes.c_void_p,ctypes.c_void_p]
def ptrace(request,pid,data=0):
 result=libc.ptrace(request,pid,None,data)
 if result==-1:
  number=ctypes.get_errno();raise OSError(number,os.strerror(number))
 return result

def threads(pid):
 result={}
 for path in (Path('/proc')/str(pid)/'task').iterdir():
  if path.name.isdigit():result[int(path.name)]=(path/'stat').read_text().rsplit(')',1)[1].split()[0]
 return result

@contextmanager
def frozen(pid,receipt):
 fd=os.pidfd_open(pid);seized=False;signalled=False;started=time.monotonic()
 try:
  assert all(s not in ('T','t','Z') for s in threads(pid).values()),'unexpected existing stop'
  signal.pidfd_send_signal(fd,signal.SIGSTOP);signalled=True
  deadline=time.monotonic()+2
  while not all(s=='T' for s in threads(pid).values()):
   if time.monotonic()>deadline:raise TimeoutError('threads did not stop')
   time.sleep(.002)
  ptrace(0x4206,pid);seized=True
  ptrace(0x4207,pid)
  deadline=time.monotonic()+2
  while True:
   who,status=os.waitpid(pid,os.WNOHANG|0x40000000)
   if who:break
   if time.monotonic()>deadline:raise TimeoutError('ptrace interrupt did not stop')
   time.sleep(.002)
  assert os.WIFSTOPPED(status)
  regs=Registers();ptrace(12,pid,ctypes.byref(regs))
  receipt.update(registers={n:hex(getattr(regs,n)) for n,_ in regs._fields_},wait_status=status,stopped_threads=threads(pid))
  yield regs
 finally:
  try:
   if seized:ptrace(17,pid)
  finally:
   if signalled:signal.pidfd_send_signal(fd,signal.SIGCONT)
   os.close(fd)
   receipt['elapsed_seconds']=time.monotonic()-started

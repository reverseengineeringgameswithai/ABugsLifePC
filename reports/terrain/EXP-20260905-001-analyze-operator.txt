    assert w['bytes_written'] == w['bytes'] == p['bytes']
    assert int(w['address'], 16) == int(plan['address'], 16) + p['offset']
    assert w['before_prefix_sha256'] == plan['files'][p['from']]['sha256']
    assert w['target_prefix_sha256'] == w['after_prefix_sha256'] == plan['files'][p['to']]['sha256']
    assert w['frame_end_gate'] and w['renderer_return'] == '0x44dbf3'

# Exactly the EXP-010/011 selection, frozen before this live run.
prior_path = HERE.parent / 'EXP-20260904-009/pixel-depths-exploratory.json'
prior = load(prior_path)
camera = plan['state']['camera_translation']
a, b = 1409 / 4096, 3845 / 4096
den = a * a + b * b
analytic = []
for row in prior['rows']:
    sx, sy = row['xy']
    z = (-108544 - camera[1]) / 32 * 320 / (sy - 240)
    x = (sx - 320) / 320 * z
    wx = camera[0] + 32 * (a * x - b * z) / den
    wz = camera[2] + 32 * (b * x + a * z) / den
    actor = row['depths_actor_coarse_split'][0]
    if 163200 <= wx <= 184000 and 497600 <= wz <= 520000 and actor is not None and z / 16384 < actor:
        analytic.append(tuple(row['xy']))
inversions = [tuple(r['xy']) for r in prior['inversions']]
residuals = [(342, 315), (344, 315), (298, 319), (299, 319), (300, 319), (301, 319)]
assert len(set(analytic)) == len(analytic) == 259
assert len(set(inversions)) == len(inversions) == 255
assert set(residuals) <= set(analytic)
blue = lambda rgb: rgb[2] > rgb[0] + 6 and rgb[2] > rgb[1] + 2
roi = (285, 285, 350, 340)
images, raw = {}, {}
assert list(result['captures']) == plan['capture_labels']
for arm, rows in result['captures'].items():
    assert len(rows) == 3
    images[arm] = []
    for row in rows:
        assert row['state'] == plan['state']
        assert digest(Path(row['frame'])) == row['sha256']
        with Image.open(row['frame']) as im:
            assert im.size == (640, 480)
            image = im.convert('RGB')
        assert any(low != high for low, high in image.getextrema())
        images[arm].append(image)
    raw[arm] = [im.crop(roi).tobytes() for im in images[arm]]
    assert len(set(raw[arm])) == 1, ('unstable ROI', arm)
assert raw['original_before'] == raw['original_restored']
assert raw['control_before'] == raw['control_restored']

counts = {name: {arm: [sum(blue(im.getpixel(xy)) for xy in points) for im in ims]
                         for arm, ims in images.items()}
          for name, points in [('analytic259', analytic), ('residual6', residuals), ('prior255', inversions)]}
# All expectations are from the existing offline plan, not a revised mask.
assert list(counts['analytic259'].values()) == [[259]*3, [6]*3, [0]*3, [6]*3, [259]*3]
assert list(counts['residual6'].values()) == [[6]*3, [6]*3, [0]*3, [6]*3, [6]*3]
assert list(counts['prior255'].values()) == [[255]*3, [0]*3, [0]*3, [0]*3, [255]*3]

old = load(OLD / 'result.json')
old_matches = {}
for current, previous in [('original_before', 'before'), ('control_before', 'candidate'), ('original_restored', 'restored')]:
    old_rois = []
    for row in old['captures'][previous]:
        assert digest(Path(row['frame'])) == row['sha256'] and row['state'] == plan['state']
        with Image.open(row['frame']) as im:
            old_rois.append(im.convert('RGB').crop(roi).tobytes())
    old_matches[current] = raw[current] == old_rois

comparison = {
    'roi': roi, 'roi_pixels': 3575, 'state_identical_all_15': True,
    'triplet_roi_identical_each_arm': True, 'original_restored_roi_identical': True,
    'control_restored_roi_identical': True, 'blue_filter': 'B > R + 6 and B > G + 2',
    'blue_counts': counts,
    'selected_residual_rgb': [{'xy': xy, 'rgb_by_arm': {arm: [im.getpixel(xy) for im in ims] for arm, ims in images.items()}} for xy in residuals],
    'rgb_roi_sha256': {arm: [hashlib.sha256(b).hexdigest() for b in values] for arm, values in raw.items()},
    'EXP010_roi_matches': old_matches,
    'control_to_alternate_changed_roi_pixels': sum(images['control_before'][0].getpixel((x,y)) != images['alternate'][0].getpixel((x,y)) for y in range(roi[1],roi[3]) for x in range(roi[0],roi[2])),
    'scope': 'Frozen EXP009/010/011 pixel selections. ROI also contains legitimately exposed model and textured box pixels; their total difference does not measure defects. No full-frame equality or moving-camera acceptance.'
}
(HERE / 'comparison.json').write_text(json.dumps(comparison, indent=2) + '\n')
